**冒泡排序**
```python
def sortArray(self, nums: List[int]) -> List[int]:
  for i in range(len(nums) - 1):
      for j in range(i, len(nums)):
          if nums[i] > nums[j]:
              nums[i], nums[j] = nums[j], nums[i]
  return nums
```

**选择排序**
```python
def sortArray(self, nums: List[int]) -> List[int]:
    for i in range(len(nums) - 1):
        min_index = i
        for j in range(i, len(nums)):
            if nums[min_index] > nums[j]:
                min_index = j
        nums[i], nums[min_index] = nums[min_index], nums[i]
    return nums
```

```python
class Solution:
    def sortArray(self, nums: List[int]) -> List[int]:
        def quick_sort(arr, low, high):  # low high是当前数组范围
            if low >= high:
                return
            pivot_idx = random.randint(low, high)
            arr[low], arr[pivot_idx] = arr[pivot_idx], arr[low]
            pivot = arr[low]  # 记录pivot值，left位置空出来

            left, right = low, high  # left, right是当前需要使用的指针
            while left < right:
                # 从right找<=的数，放入left（全相等时也移动
                while left < right and arr[right] > pivot:
                    right -= 1
                # 如果没遍历完
                if left < right:
                    arr[left] = arr[right]
                    left += 1  # 指向下一个待排序元素

                while left < right and arr[left] < pivot:
                    left += 1
                if left < right:
                    arr[right] = arr[left]
                    right -= 1
            # 最后pivot放入left=right位置
            arr[left] = pivot
            mid = left
            quick_sort(arr, low, mid - 1)
            quick_sort(arr, mid + 1, high)
        quick_sort(nums, 0, len(nums) - 1)
        return nums
```


**归并排序**
```python
class Solution:
    def sortArray(self, nums: List[int]) -> List[int]:
        def mergeSort(arr, low, high):
            if low >= high:
                return
            mid = low + (high - low) // 2
            mergeSort(arr, low, mid)
            mergeSort(arr, mid + 1, high)

            left, right = low, mid + 1  # 两个子序列的起点
            tmp = []
            while left <= mid and right <= high:  # 未到终点
                if arr[left] <= arr[right]:  # left的小
                    tmp.append(arr[left])
                    left += 1
                else:
                    tmp.append(arr[right])
                    right += 1
            while left <= mid:  # 剩余部分加入tmp
                tmp.append(arr[left])
                left += 1
            while right <= high:
                tmp.append(arr[right])
                right += 1
            arr[low: high + 1] = tmp  # 存入对应位置
        mergeSort(nums, 0, len(nums) - 1)
        return nums
```
